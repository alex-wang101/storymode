# Dependencies for the videofind pipeline.
# Auto-installed by main.py if any core import is missing.

# pipeline/sponsor_detect.py  --  Xenova/sponsorblock-small (T5 seq2seq) + tokenizer
torch
transformers

# pipeline/transcript_ingest.py  --  fetch YouTube auto-captions
yt-dlp

# pipeline/frame_sampler.py  --  decode video frames with OpenCV
opencv-python

# pipeline/object_detector.py  --  OWLv2 text-guided detection

easyocr
rapidfuzz
pillow

# pipeline/schema.py  --  numpy.ndarray field on SampledFrame

# pinning to <2 keeps the full stack consistent.
numpy<2

torchvision

uvicorn[standard]

python-multipart

# api/reference_gen.py  --  cloud-VLM providers (key-prefix routed).
anthropic
openai
google-genai

# optimum + auto-gptq are needed by transformers to load GPTQ-Int8 weights.
